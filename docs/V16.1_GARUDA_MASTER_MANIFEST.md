```

├── @home        → /home          (user data)
├── @opt         → /opt           (SCARLIX + Docker configs)
├── @var         → /var           (Docker images, logs, pacman cache)
├── @models      → /models        (AI modely — safetensors + GGUF)

├── @files       → /mnt/files     (média, Nextcloud)

└── @backup      → /mnt/backup    (Restic backups)

Snapper konfigurácia:
- Automatický snapshot každú hodinu (timeline)
- Snapshot pred každou `pacman` operáciou
- Rollback: `snapper rollback` (BTRFS atomic, <5s)
```

### 2.7 Snapper konfigurácia (automatické snapshoty)

```bash
# Garuda už má Snapper nakonfigurovaný, ale pridaj SCARLIX subvolumes:
sudo snapper -c scarlix-opt create-config /opt
sudo snapper -c scarlix-models create-config /models
sudo snapper -c scarlix-var create-config /var

# Timeline snapshots (každú hodinu):
sudo snapper -c root set-config "TIMELINE_CREATE=yes TIMELINE_CLEANUP=yes"
sudo snapper -c scarlix-opt set-config "TIMELINE_CREATE=yes TIMELINE_CLEANUP=yes"
sudo snapper -c scarlix-models set-config "TIMELINE_CREATE=yes TIMELINE_CLEANUP=yes"

# Pre-pacman snapshot (pred každou aktualizáciou):
sudo tee /etc/pacman.d/hooks/snap-pre.hook << 'EOF'
[Trigger]
Operation = Upgrade
Operation = Install
Operation = Remove
Type = Package
Target = *
[Action]
Description = Creating BTRFS snapshot before pacman transaction
Depends = snapper
When = PreTransaction
Exec = /usr/bin/snapper -c root create --description "pre-pacman"
EOF
```

### 3.4 Stabilné GPU indexy (udev rules)

```bash
GPU_PCI=$(lspci | grep -i nvidia | awk '{print $1}')
GPU0_PCI=$(echo "$GPU_PCI" | head -1)
GPU1_PCI=$(echo "$GPU_PCI" | tail -1)
sudo tee /etc/udev/rules.d/99-nvidia-stable.rules << EOF
KERNEL=="nvidia", KERNELS=="0000:$GPU0_PCI", SYMLINK+="nvidia-ai"
KERNEL=="nvidia_modeset", KERNELS=="0000:$GPU0_PCI", SYMLINK+="nvidia-ai-modeset"
KERNEL=="nvidia_uvm", KERNELS=="0000:$GPU0_PCI", SYMLINK+="nvidia-ai-uvm"
KERNEL=="nvidia", KERNELS=="0000:$GPU1_PCI", SYMLINK+="nvidia-game"
KERNEL=="nvidia_modeset", KERNELS=="0000:$GPU1_PCI", SYMLINK+="nvidia-game-modeset"
KERNEL=="nvidia_uvm", KERNELS=="0000:$GPU1_PCI", SYMLINK+="nvidia-game-uvm"
```

### scarlix-mode v16.1 (Garuda adaptácia)

```bash
#!/usr/bin/env bash
# /usr/local/bin/scarlix-mode
set -euo pipefail
LOG_FILE="/var/log/scarlix-mode.log"
STATE_FILE="/var/lib/scarlix/current-mode"
LOCK_FILE="/tmp/scarlix-mode.lock"

log() { echo "[$(date '+%Y-%m-%d %H:%M:%S')] $1" | tee -a "$LOG_FILE"; }

exec 200>"$LOCK_FILE"
flock -n 200 || { log "ERROR: iny proces prepina rezim"; exit 1; }

MODE="${1:-status}"
CURRENT_MODE=$(cat "$STATE_FILE" 2>/dev/null || echo "unknown")

dump_vram() {
  log "Uvolnujem VRAM..."
  curl -s -X POST "http://localhost:11434/api/generate" -d '{"model":"qwen3.6:14b","keep_alive":0}' >/dev/null 2>&1 || true
  curl -s -X POST "http://localhost:11435/api/generate" -d '{"model":"qwen3.6:14b","keep_alive":0}' >/dev/null 2>&1 || true
  docker stop sglang 2>/dev/null || true
  sleep 3
}

case "$MODE" in
  ai)
    [[ "$CURRENT_MODE" == "ai" ]] && { log "Uz v AI mode."; exit 0; }
    log "=== AI mode ==="
    # Garuda gaming: zatvor Steam/Lutris
    killall steam lutris 2>/dev/null || true
    sudo systemctl stop sunshine 2>/dev/null || true
    docker compose -f /opt/scarlix/ai/comfyui/docker-compose.yml --profile creative stop 2>/dev/null || true
    docker compose -f /opt/scarlix/ai/video/docker-compose.yml --profile creative stop 2>/dev/null || true
    docker compose -f /opt/scarlix/ai/musicgen/docker-compose.yml --profile creative stop 2>/dev/null || true
    docker compose -f /opt/scarlix/ai/sglang/docker-compose.yml up -d
    echo "ai" | tee "$STATE_FILE" >/dev/null
    ;;
  game)
    [[ "$CURRENT_MODE" == "game" ]] && { log "Uz v Game mode."; exit 0; }
    log "=== GAME mode ==="
    dump_vram
    docker compose -f /opt/scarlix/ai/comfyui/docker-compose.yml --profile creative stop 2>/dev/null || true
    # Sunshine je NATIVNY (systemd), nie Docker:
    sudo systemctl start sunshine
    echo "game" | tee "$STATE_FILE" >/dev/null
    ;;
  creative)
    [[ "$CURRENT_MODE" == "creative" ]] && { log "Uz v Creative mode."; exit 0; }
    log "=== CREATIVE mode ==="
    dump_vram
    sudo systemctl stop sunshine 2>/dev/null || true
    docker compose -f /opt/scarlix/ai/comfyui/docker-compose.yml --profile creative up -d
    docker compose -f /opt/scarlix/ai/video/docker-compose.yml --profile creative up -d
    docker compose -f /opt/scarlix/ai/musicgen/docker-compose.yml --profile creative up -d
    echo "creative" | tee "$STATE_FILE" >/dev/null
    ;;
  turbo)
    log "=== TURBO mode ==="
    docker compose -f /opt/scarlix/ai/sglang/docker-compose.yml up -d
    docker compose -f /opt/scarlix/ai/ollama/docker-compose-main.yml up -d
    echo "turbo" | tee "$STATE_FILE" >/dev/null
    ;;
  offline)
    log "=== OFFLINE mode ==="
    dump_vram
    sudo systemctl stop sunshine 2>/dev/null || true
    docker compose -f /opt/scarlix/ai/llamacpp/docker-compose.yml up -d
    echo "offline" | tee "$STATE_FILE" >/dev/null
    ;;
  tv)
    log "=== TV mode ==="
    sudo systemctl start sunshine
    sudo systemctl start scarlix-tv-mode.service
    echo "tv" | tee "$STATE_FILE" >/dev/null
    ;;
  status)
    echo "=== SCARLIX OS v16.1 Status ==="
    echo "Mode: $CURRENT_MODE"
    echo "Base: Garuda Linux $(cat /etc/garuda-release 2>/dev/null || echo 'Arch')"
    echo "Kernel: $(uname -r)"
    nvidia-smi --query-gpu=index,name,memory.used,memory.free,utilization.gpu --format=csv,noheader 2>/dev/null || echo "nvidia-smi not available"
    echo ""
    echo "BTRFS snapshots:"
    sudo snapper -c root list 2>/dev/null | tail -5
    echo ""
    docker ps --format "table {{.Names}}\t{{.Status}}" | head -30
    ;;
  *)
    echo "Usage: scarlix-mode {ai|game|creative|turbo|offline|tv|status}"
    exit 1
    ;;
esac
flock -u 200
```

> **Kľúčový rozdiel oproti V15:** `scarlix-mode game` spúšťa Sunshine cez `systemctl start sunshine` (natívny systemd služba), nie Docker. Steam, RetroArch, Heroic sú už natívne nainštalované v Garude.

```bash
# installer/scripts/build-iso.sh
set -euo pipefail
VERSION="16.1.0"
WORK_DIR="/tmp/scarlix-iso-builder"
OUTPUT_DIR="$HOME/scarlix-iso"
mkdir -p "$WORK_DIR" "$OUTPUT_DIR"

echo "=== SCARLIX OS v16.1 — Garuda ISO Builder ==="

# 1. Stiahni Garuda Dr460nized ISO
GARUDA_ISO="garuda-dr460nized-linux-kde-2401-linux-zen.iso"
if [ ! -f "$WORK_DIR/$GARUDA_ISO" ]; then
  echo "[1/7] Downloading Garuda Dr460nized ISO..."
  wget -O "$WORK_DIR/$GARUDA_ISO" \
    "https://downloads.garudalinux.org/iso/garuda-dr460nized/2401/$GARUDA_ISO"
fi

# 2. Extrahuj ISO
mkdir -p "$WORK_DIR/iso-content" "$WORK_DIR/iso-mount"
sudo mount -o loop "$WORK_DIR/$GARUDA_ISO" "$WORK_DIR/iso-mount"
cp -r "$WORK_DIR/iso-mount/"* "$WORK_DIR/iso-content/"
sudo umount "$WORK_DIR/iso-mount"

# 3. Pridaj SCARLIX overlay
echo "[2/7] Adding SCARLIX overlay..."
mkdir -p "$WORK_DIR/iso-content/opt/scarlix-src"
cp -r scarlix-src/* "$WORK_DIR/iso-content/opt/scarlix-src/"
cp -r installer/* "$WORK_DIR/iso-content/opt/scarlix-installer/"

# 4. Pridaj Calamares SCARLIX konfiguráciu
echo "[3/7] Configuring Calamares installer..."
cat > "$WORK_DIR/iso-content/etc/calamares/scarlix-settings.conf" << 'EOF'
---
efiSystemPartitionSize: 512MiB
btrfsSubvolumes:
  - subvolume: /@
    name: Root
  - subvolume: /@home
    name: Home
  - subvolume: /@opt
    name: Opt
  - subvolume: /@var
    name: Var
  - subvolume: /@models
    name: Models
  - subvolume: /@games
    name: Games
defaultFileSystem: btrfs
EOF

# 5. Pridaj first-boot service
echo "[4/7] Adding first-boot service..."
cat > "$WORK_DIR/iso-content/etc/systemd/system/scarlix-first-boot.service" << 'EOF'
[Unit]
Description=SCARLIX OS v16.1 First Boot Setup
After=network-online.target
Wants=network-online.target
ConditionPathExists=!/opt/scarlix/.installed
[Service]
Type=oneshot

ExecStartPost=/bin/touch /opt/scarlix/.installed
RemainAfterExit=yes
TimeoutStartSec=3600
[Install]
WantedBy=multi-user.target
EOF

# 6. Pridaj GRUB menu entries
echo "[5/7] Configuring GRUB..."
cat > "$WORK_DIR/iso-content/boot/grub/grub.cfg" << 'GRUB'
set default=0
set timeout=15
set gfxmode=auto
set gfxpayload=keep
insmod all_video
insmod gfxterm
set menu_color_normal=white/black
set menu_color_highlight=cyan/black

menuentry "  SCARLIX OS v16.1 — Main PC (Garuda Dr460nized + AI + Gaming)" {
  set gfxpayload=keep
  linux /boot/vmlinuz-linux-zen archisobasedir=garuda archisolabel=SCARLIX_V16 scarlix_pc=main
  initrd /boot/initramfs-linux-zen.img
}

menuentry "  SCARLIX OS v16.1 — HP Agent PC (Headless Worker)" {
  set gfxpayload=keep
  linux /boot/vmlinuz-linux-zen archisobasedir=garuda archisolabel=SCARLIX_V16 scarlix_pc=hp_agent
  initrd /boot/initramfs-linux-zen.img
}

menuentry "  Garuda Linux (Original — no SCARLIX)" {
  set gfxpayload=keep
  linux /boot/vmlinuz-linux-zen archisobasedir=garuda archisolabel=GARUDA
  initrd /boot/initramfs-linux-zen.img
}
GRUB

# 7. Build ISO
echo "[6/7] Building ISO..."
sudo xorriso -as mkisofs \
  -iso-level 3 -full-iso9660-filenames \
  -volid "SCARLIX_V16" \
  -eltorito-boot boot/grub/bios.img \
  -eltorito-alt-boot -e EFI/garuda/efiboot.img -no-emul-boot \
  -isohybrid-gpt-basdat \
  -o "$OUTPUT_DIR/scarlix-os-v16.1-garuda.iso" \
  "$WORK_DIR/iso-content"

echo "[7/7] Done!"
echo "ISO: $OUTPUT_DIR/scarlix-os-v16.1-garuda.iso"
```

## ČASŤ 15 — BACKUP (BTRFS + Restic)

```bash
#!/usr/bin/env bash
# /opt/scarlix/scripts/backup.sh — BTRFS snapshot + Restic
set -euo pipefail
source /etc/scarlix/.env
TS=$(date +%Y%m%d-%H%M%S)

echo "=== SCARLIX OS v16.1 Backup ==="

# BTRFS snapshot (atomický, <1s):
echo "[1/3] Creating BTRFS snapshot..."
sudo btrfs subvolume snapshot / /mnt/backup/btrfs-snap-${TS}
echo "Done (BTRFS atomic snapshot)."

# Restic backup:
echo "[2/3] Running Restic backup..."
export RESTIC_REPOSITORY=/mnt/backup/restic
export RESTIC_PASSWORD=$RESTIC_PASSWORD
restic backup /etc/scarlix /opt/scarlix /var/lib/scarlix /var/lib/buzz \
  --exclude '*.log' \
  --tag scarlix-v16 --tag daily

# Cleanup:
echo "[3/3] Cleaning old backups..."
restic forget --keep-daily 7 --keep-weekly 4 --keep-monthly 6 --prune

# Cleanup old BTRFS snapshots (>7 days):
sudo btrfs subvolume list /mnt/backup | grep btrfs-snap | awk '{print $NF}' | while read snap; do
  SNAP_DATE=$(echo "$snap" | grep -oP '\d{8}-\d{6}')
  if [[ $(date -d "${SNAP_DATE:0:8}" +%s 2>/dev/null || echo 0) -lt $(date -d "7 days ago" +%s) ]]; then
    sudo btrfs subvolume delete "$snap" 2>/dev/null || true
  fi
done

echo "=== Backup complete: $TS ==="
```
